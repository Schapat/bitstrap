Einbinden

Drei Wege, je nach Projekt.

Zwei Dateien, kein Build.

<link rel="stylesheet" href="bitstrap.min.css">
<script src="bitstrap.min.js" defer></script>

<!-- optional -->
<script src="bitstrap-ui.min.js" defer></script>
<script src="bitstrap-forms.min.js" defer></script>

Fuer Projekte mit Bundler.

npm install bitstrap

import "bitstrap/css";
import "bitstrap";
import "bitstrap/ui";      // Drawer, Stepper, Tabellen
import "bitstrap/forms";   // Datepicker, Combobox

Nur einbinden, was gebraucht wird.

@import "bitstrap/src/00-tokens.css";   /* immer zuerst */
@import "bitstrap/src/01-base.css";
@import "bitstrap/src/04-buttons.css";
@import "bitstrap/src/06-components.css";

/* Ein eigenes Theme - nur Variablen */
.bit-theme-meins {
  --bit-bg:      #1a1c2c;
  --bit-surface: #29366f;
  --bit-text:    #f4f4f4;
  --bit-border:  #f4f4f4;

  --bit-primary:    #ef7d57;
  --bit-primary-fg: #1a1c2c;
  --bit-success:    #a7f070;
  --bit-danger:     #b13e53;
}

/* Und die Groesse? Eine Variable. */
:root { --bit-px: 6px; }
